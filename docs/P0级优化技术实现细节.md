### 2. 目标元件库映射构建

```rust
async fn build_target_cata_map(
    refnos: &[RefnoEnum]
) -> Result<Arc<DashMap<String, CataHashRefnoKV>>> {
    let target_cata_map = Arc::new(DashMap::new());
    
    for refno in refnos {
        // 查询元件库哈希
        if let Some(cata_hash) = aios_core::get_cata_hash(*refno).await? {
            // 构建或更新元件库映射
            target_cata_map.entry(cata_hash.clone())
                .and_modify(|entry| {
                    entry.group_refnos.push(*refno);
                })
                .or_insert(CataHashRefnoKV {
                    cata_hash: cata_hash.clone(),
                    group_refnos: vec![*refno],
                    exist_inst: false,
                    ptset: None,
                });
        }
    }
    
    Ok(target_cata_map)
}
```

### 3. 并行处理核心逻辑

```rust
async fn process_catalogs_parallel(
    target_cata_map: &Arc<DashMap<String, CataHashRefnoKV>>
) -> Result<ShapeInstancesData> {
    // 动态获取CPU核心数
    let cpu_count = std::thread::available_parallelism()
        .map(|n| n.get())
        .unwrap_or(4);
    
    // 创建信号量控制并发
    let semaphore = Arc::new(Semaphore::new(cpu_count));
    let shape_insts_data = Arc::new(Mutex::new(ShapeInstancesData::default()));
    
    // 收集所有唯一的元件库哈希
    let unique_cata_hashes: Vec<String> = target_cata_map
        .iter()
        .map(|entry| entry.key().clone())
        .collect();
    
    info!("🔄 开始并行处理 {} 个唯一元件库", unique_cata_hashes.len());
    
    // 创建并行任务
    let tasks: Vec<_> = unique_cata_hashes
        .into_iter()
        .map(|cata_hash| {
            let semaphore = semaphore.clone();
            let target_cata_map = target_cata_map.clone();
            let shape_insts_data = shape_insts_data.clone();
            
            tokio::spawn(async move {
                // 获取信号量许可
                let _permit = semaphore.acquire().await.unwrap();
                
                // 处理单个元件库
                process_single_catalog(
                    &cata_hash,
                    &target_cata_map,
                    &shape_insts_data
                ).await
            })
        })
        .collect();
    
    // 等待所有任务完成
    let results = join_all(tasks).await;
    
    // 检查错误
    for result in results {
        if let Err(e) = result {
            error!("并行任务执行失败: {}", e);
        }
    }
    
    // 返回结果
    let final_data = shape_insts_data.lock().await;
    Ok(final_data.clone())
}
```

### 4. 单个元件库处理

```rust
async fn process_single_catalog(
    cata_hash: &str,
    target_cata_map: &Arc<DashMap<String, CataHashRefnoKV>>,
    shape_insts_data: &Arc<Mutex<ShapeInstancesData>>
) -> Result<()> {
    let cata_start = Instant::now();
    
    // 查找使用此元件库的所有元素
    let related_elements: Vec<_> = target_cata_map
        .iter()
        .filter(|entry| &entry.cata_hash == cata_hash)
        .flat_map(|entry| entry.group_refnos.clone())
        .collect();
    
    debug!("📦 处理元件库 {} (关联元素: {})", cata_hash, related_elements.len());
    
    // 处理每个使用此元件库的元素
    for refno in related_elements {
        // 获取元素属性
        let ele_att = aios_core::get_named_attmap(refno).await?;
        
        // 获取世界变换
        let origin_trans = aios_core::get_world_transform(refno).await
            .unwrap_or(Transform::IDENTITY);
        
        // 获取元件库参考号
        let cat_refno = aios_core::get_cat_refno(refno).await?
            .ok_or_else(|| anyhow!("元素 {} 没有元件库参考号", refno))?;
        
        // 生成几何体
        gen_cata_single_geoms(
            refno,
            &Default::default(), // brep_shape_map
            &Default::default(), // design_axis_map
        ).await?;
        
        // 获取通用类型
        let generic_type = get_generic_type(refno).await.unwrap_or_default();
        
        // 创建几何体信息
        let geos_info = EleGeosInfo {
            refno,
            ele_att,
            origin_trans,
            cat_refno,
            generic_type,
            // ... 其他字段
        };
        
        // 线程安全地添加到结果中
        {
            let mut data = shape_insts_data.lock().await;
            data.inst_info_map.insert(refno, geos_info);
        }
    }
    
    let cata_time = cata_start.elapsed();
    if cata_time.as_millis() > 50 {
        debug!("⏱️ 元件库 {} 处理耗时: {}ms", cata_hash, cata_time.as_millis());
    }
    
    Ok(())
}
```

```rust
impl ParallelCataPerformanceStats {
    pub fn calculate_metrics(&self) -> (f64, f64, f64, f64) {
        let total_secs = self.total_time.as_secs_f64();
        
        // 元件库处理速度 (元件库/秒)
        let cata_speed = if total_secs > 0.0 {
            self.total_cata_count as f64 / total_secs
        } else { 0.0 };
        
        // 实例生成速度 (实例/秒)
        let instance_speed = if total_secs > 0.0 {
            self.generated_instances as f64 / total_secs
        } else { 0.0 };
        
        // 平均元件库处理时间 (ms/元件库)
        let avg_cata_time = if self.total_cata_count > 0 {
            self.parallel_processing_time.as_millis() as f64 / self.total_cata_count as f64
        } else { 0.0 };
        
        // 平均实例生成时间 (ms/实例)
        let avg_instance_time = if self.generated_instances > 0 {
            self.total_time.as_millis() as f64 / self.generated_instances as f64
        } else { 0.0 };
        
        (cata_speed, instance_speed, avg_cata_time, avg_instance_time)
    }
    
    pub fn efficiency_score(&self) -> f64 {
        if self.success { 100.0 } else { 0.0 }
    }
    
    pub fn efficiency_rating(&self) -> &'static str {
        let score = self.efficiency_score();
        match score {
            90.0..=100.0 => "优秀 🌟",
            70.0..=89.9 => "良好 👍",
            50.0..=69.9 => "一般 ⚡",
            _ => "需要改进 ⚠️"
        }
    }
}
```

## 🔧 最佳实践

### 1. 错误处理策略
- **优雅降级**：单个元件库失败不影响整体处理
- **详细日志**：记录每个处理步骤的详细信息
- **资源清理**：确保异常情况下资源正确释放

### 2. 内存管理
- **流式处理**：避免一次性加载大量数据
- **及时释放**：处理完成后立即释放不需要的资源
- **共享数据**：使用Arc减少数据拷贝

### 3. 并发控制
- **动态调整**：根据系统资源动态调整并发度
- **负载均衡**：确保任务均匀分布到各个线程
- **死锁避免**：合理的锁顺序和超时机制
